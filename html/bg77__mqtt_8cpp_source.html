204void Bg77Mqtt::handleModemLine(const String &line)
205{
206 if (line.length() == 0)
207 {
208 return;
209 }
210
211 _pc.print("[MODEM] ");
212 _pc.println(line);
213
214 if (line.indexOf("+CEREG:") >= 0)
215 {
216 int comma = line.indexOf(',');
217 bool registered = false;
218
219 if (comma >= 0)
220 {
221 char stat = line.charAt(comma + 1);
222 if (stat == '1' || stat == '5')
223 {
224 char after = (comma + 2 < (int)line.length()) ? line.charAt(comma + 2) : '\n';
225 if (after == ',' || after == '\r' || after == '\n' || after == '\0' || after == ' ')
226 {
227 registered = true;
228 }
229 }
230 }
231
232 _networkRegistered = registered;
233 printStateSnapshot("CEREG");
234 }
235
236 if (line.indexOf("+CGATT: 1") >= 0)
237 {
238 _packetAttached = true;
239 printStateSnapshot("CGATT");
240 }
241 else if (line.indexOf("+CGATT: 0") >= 0)
242 {
243 _packetAttached = false;
244 printStateSnapshot("CGATT");
245 }
246
247 if (line.indexOf("+QIURC: \"pdpdeact\",1") >= 0)
248 {
249 _packetAttached = false;
250 _mqttSocketOpen = false;
251 _mqttConnected = false;
252 logWarn("PDP deactivated");
253 printStateSnapshot("PDPDEACT");
254 }
255
256 if (line.indexOf("+QMTOPEN: 0,0") >= 0)
257 {
258 _mqttSocketOpen = true;
259 printStateSnapshot("QMTOPEN OK");
260 }
261 else if (line.startsWith("+QMTOPEN: 0,"))
262 {
263 _mqttSocketOpen = false;
264 _mqttConnected = false;
265 printStateSnapshot("QMTOPEN FAIL");
266 }
267
268 if (line.indexOf("+QMTCONN: 0,0,0") >= 0)
269 {
270 _mqttConnected = true;
271 _mqttReconnectFailures = 0;
272 _lastMqttOkMs = millis();
273 printStateSnapshot("QMTCONN OK");
274 }
275 else if (line.startsWith("+QMTCONN: 0,"))
276 {
277 _mqttConnected = false;
278 printStateSnapshot("QMTCONN FAIL");
279 }
280
281 if (line.startsWith("+QMTSTAT:"))
282 {
283 _mqttConnected = false;
284 printStateSnapshot("QMTSTAT");
285 }
286
287 if (line.startsWith("+QMTCLOSE:"))
288 {
289 _mqttSocketOpen = false;
290 _mqttConnected = false;
291 printStateSnapshot("QMTCLOSE");
292 }
293
294 if (line.startsWith("+QMTDISC:"))
295 {
296 _mqttConnected = false;
297 printStateSnapshot("QMTDISC");
298 }
299
300 if (line.startsWith("+QMTRECV:"))
301 {
302 int firstComma = line.indexOf(',');
303 int secondComma = (firstComma >= 0) ? line.indexOf(',', firstComma + 1) : -1;
304 int topicOpen = (secondComma >= 0) ? line.indexOf('"', secondComma + 1) : -1;
305 int topicClose = (topicOpen >= 0) ? line.indexOf('"', topicOpen + 1) : -1;
306 int payloadOpen = (topicClose >= 0) ? line.indexOf('"', topicClose + 1) : -1;
307 int payloadClose = line.lastIndexOf('"');
308
309 if (payloadOpen >= 0 && payloadClose > payloadOpen)
310 {
311 if (_mqttQueueCount < MQTT_QUEUE_SIZE)
312 {
313 _mqttQueue[_mqttQueueTail] = line.substring(payloadOpen + 1, payloadClose);
314 _mqttQueueTail = (_mqttQueueTail + 1) % MQTT_QUEUE_SIZE;
315 _mqttQueueCount++;
316 }
317 else
318 {
319 logWarn("MQTT queue pleine, message perdu");
320 }
321 }
322 }

1027uint32_t Bg77Mqtt::reconnectBackoffMs() const
1028{
1029 if (_mqttReconnectFailures == 0)
1030 {
1031 return MQTT_RETRY_DELAY_MS;
1032 }
1033
1034 uint32_t backoff = MQTT_RETRY_DELAY_MS;
1035
1036 for (uint8_t i = 0; i < _mqttReconnectFailures; i++)
1037 {
1038 backoff *= 2UL;
1039 if (backoff > 120000UL)
1040 {
1041 backoff = 120000UL;
1042 break;
1043 }
1044 }
1045
1046 return backoff;

1139String Bg77Mqtt::convertQltsToIso8601(const String &qlts) const
1140{
1141 // Expected example:
1142 // 2026/04/20,20:04:24+08,0
1143 if (qlts.length() < 22)
1144 {
1145 return "";
1146 }
1147
1148 int slash1 = qlts.indexOf('/');
1149 int slash2 = qlts.indexOf('/', slash1 + 1);
1150 int comma = qlts.indexOf(',');
1151 int colon1 = qlts.indexOf(':', comma + 1);
1152 int colon2 = qlts.indexOf(':', colon1 + 1);
1153
1154 if (slash1 < 0 || slash2 < 0 || comma < 0 || colon1 < 0 || colon2 < 0)
1155 {
1156 return "";
1157 }
1158
1159 String year = qlts.substring(0, slash1);
1160 String month = qlts.substring(slash1 + 1, slash2);
1161 String day = qlts.substring(slash2 + 1, comma);
1162
1163 String hour = qlts.substring(comma + 1, colon1);
1164 String minute = qlts.substring(colon1 + 1, colon2);
1165 String second = qlts.substring(colon2 + 1, colon2 + 3);
1166
1167 int signPosPlus = qlts.indexOf('+', colon2 + 1);
1168 int signPosMinus = qlts.indexOf('-', colon2 + 1);
1169 int signPos = -1;
1170
1171 if (signPosPlus >= 0)
1172 {
1173 signPos = signPosPlus;
1174 }
1175 else if (signPosMinus >= 0)
1176 {
1177 signPos = signPosMinus;
1178 }
1179
1180 String tz = "+00:00";
1181
1182 if (signPos >= 0 && signPos + 3 <= (int)qlts.length())
1183 {
1184 char sign = qlts.charAt(signPos);
1185 int quarterHours = qlts.substring(signPos + 1, signPos + 3).toInt();
1186 int totalMinutes = quarterHours * 15;
1187
1188 int hours = totalMinutes / 60;
1189 int minutes = totalMinutes % 60;
1190
1191 char buffer[16];
1192 snprintf(buffer, sizeof(buffer), "%c%02d:%02d", sign, hours, minutes);
1193 tz = String(buffer);
1194 }
1195
1196 String iso = "";
1197 iso += year;
1198 iso += "-";
1199 iso += month;
1200 iso += "-";
1201 iso += day;
1202 iso += "T";
1203 iso += hour;
1204 iso += ":";
1205 iso += minute;
1206 iso += ":";
1207 iso += second;
1208 iso += tz;
1209
1210 return iso;
1211}
