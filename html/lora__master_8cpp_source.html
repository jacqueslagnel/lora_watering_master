192 }
193 return;
194 }

205bool LoRaMaster::handleMqttCommand(const String &payload)
206{
207 String masterVerb;
208 String masterArg;
209
210 if (parseMqttMasterCommand(payload, masterVerb, masterArg))
211 {
212
213 if (masterVerb == "ota")
214 {
215 uint32_t sec = BleOta::sanitizeDurationSec((uint32_t)masterArg.toInt());
216 publishMasterOtaResult(true, sec);
217 delay(100);
218 BleOta::requestBoot(sec);
219 NVIC_SystemReset();
220 return true;
221 }
222
223 if (masterVerb == "list")
224 {
225 publishMasterListResult(true);
226 dumpNodeListTsv();
227 return true;
228 }
229
230 if (masterVerb == "schedule")
231 {
232 uint32_t sec = (uint32_t)masterArg.toInt();
233 if (sec == 0)
234 {
235 publishMasterScheduleResult(false, 0);
236 return false;
237 }
238
239 setBatterySchedulePeriodSec(sec);
240 publishMasterScheduleResult(true, sec);
241 return true;
242 }
243
244 if (masterVerb == "bat" || masterVerb == "battery")
245 {
246 if (_masterBatteryReader == nullptr)
247 {
248 publishPayload("M;bat;0;no_reader");
249 return false;
250 }
251
252 uint16_t mv = _masterBatteryReader();
253
254 String response = "M;bat;1;";
255 response += String(mv);
256
257 publishPayload(response);
258 return true;
259 }
260
261 if (masterVerb == "reboot")
262 {
263 if (_masterRebootHandler == nullptr)
264 {
265 publishPayload("M;reboot;0;no_handler");
266 return false;
267 }
268
269 publishPayload("M;reboot;1");
270 delay(500);
271 _masterRebootHandler();
272 return true;
273 }
274
275 String response = "M;error;unknown_cmd;";
276 response += masterVerb;
277 publishPayload(response);
278 return false;
279 }
280
281 uint8_t nodeId = 0;
282 uint8_t cmdCode = CMD_NONE;
283 uint16_t argU16 = 0;
284 String textArg;
285 String verb;
286 String argText;
287
288 if (!parseMqttNodeCommand(payload, nodeId, cmdCode, argU16, textArg, verb, argText))
289 {
290 if (nodeId > 0)
291 {
292 publishQueuedCommandFailure(nodeId, cmdCode, argU16);
293 }
294 else
295 {
296 publishInvalidNodeCommandResult(payload);
297 }
298 return false;
299 }
300
301 if (nodeId == 0 || nodeId > MAX_NODES)
302 {
303 publishQueuedCommandFailure(nodeId, cmdCode, argU16);
304 return false;
305 }
306
307 if (hasQueuedCommand(nodeId))
308 {
309 publishQueuedCommandFailure(nodeId, cmdCode, argU16);
310 return false;
311 }
312
313 if (!queueNodeCommand(nodeId, cmdCode, argU16, textArg.c_str(), verb.c_str(), argText.c_str()))
314 {
315 publishQueuedCommandFailure(nodeId, cmdCode, argU16);
316 return false;
317 }
318
319 return true;
320}

382uint16_t LoRaMaster::crc16Ccitt(const uint8_t *data, uint16_t len) const
383{
384 uint16_t crc = 0xFFFF;
385 for (uint16_t i = 0; i < len; i++)
386 {
387 crc ^= (uint16_t)data[i] << 8;
388 for (uint8_t b = 0; b < 8; b++)
389 {
390 if (crc & 0x8000)
391 {
392 crc = (uint16_t)((crc << 1) ^ 0x1021);
393 }
394 else
395 {
396 crc = (uint16_t)(crc << 1);
397 }
398 }
399 }
400 return crc;
401}

408uint8_t LoRaMaster::buildFrame(uint8_t type, uint8_t nodeId, uint8_t seq, const uint8_t *payload, uint8_t len)
409{
410 _txBuffer[0] = type;
411 _txBuffer[1] = nodeId;
412 _txBuffer[2] = seq;
413 _txBuffer[3] = len;
414
415 for (uint8_t i = 0; i < len; i++)
416 {
417 _txBuffer[4 + i] = payload[i];
418 }
419
420 uint16_t crc = crc16Ccitt(_txBuffer, (uint16_t)(4 + len));
421 _txBuffer[4 + len] = (uint8_t)(crc & 0xFF);
422 _txBuffer[5 + len] = (uint8_t)((crc >> 8) & 0xFF);
423
424 return getFrameTotalLength(len);

428{
429 if (_rxSize < 6)
430 {

432 }
433
434 uint8_t payloadLen = _rxBuffer[3];
435 uint16_t expectedTotal = (uint16_t)(4 + payloadLen + 2);

437 {
438 return false;
439 }
440

442 ((uint16_t)_rxBuffer[5 + payloadLen] << 8);

628void LoRaMaster::refreshOnlineStates()
629{
630 uint32_t now = millis();
631
632 for (uint8_t i = 0; i < MAX_NODES; i++)
633 {
634 if (!_nodes[i].used)
635 {
636 continue;
637 }
638
639 bool isNowOnline = ((now - _nodes[i].lastSeenMs) <= NODE_ONLINE_TIMEOUT_MS);
640
641 if (_nodes[i].online && !isNowOnline)
642 {
643 _nodes[i].online = false;
644
645 PendingNodeCommand *queued = getQueuedCommand(_nodes[i].nodeId);
646 if (queued != nullptr && queued->active)
647 {
648 printTsv("CMD_DROP", _nodes[i].nodeId, 0, 0, 0, 0, 0, queued->cmdCode, "node_offline");
649 publishQueuedCommandFailure(_nodes[i].nodeId, queued->cmdCode, queued->argU16);
650 clearQueuedCommand(_nodes[i].nodeId);
651 }
652
653 if (_schedule.cycleActive)
654 {
655 _nodes[i].lastCycleIdHandled = _schedule.cycleId;
656 }
657
658 if (!_nodes[i].notifiedLost)
659 {
660 publishNodePresence(_nodes[i].nodeId, false);
661 _nodes[i].notifiedLost = true;
662 _nodes[i].notifiedOnline = false;
663 }
664 }
665 else if (!_nodes[i].online && isNowOnline)
666 {
667 _nodes[i].online = true;
668 if (!_nodes[i].notifiedOnline)
669 {
670 publishNodePresence(_nodes[i].nodeId, true);
671 _nodes[i].notifiedOnline = true;
672 _nodes[i].notifiedLost = false;
673 }
674 }
675 }
676
677 if (_schedule.cycleActive && isScheduleCycleComplete())
678 {
679 _schedule.cycleActive = false;
680 _schedule.nextDueMs = millis() + _schedule.periodMs;
681 printTsv("SCHED_CYCLE_DONE", 0, 0, 0, 0, 0, 0, CMD_BATTERY, "battery_all_nodes");
682 }

914void LoRaMaster::sendCmdNow(uint8_t nodeId, uint8_t seq, uint8_t cmdCode, uint16_t argU16, const char *textPayload)
915{
916 uint8_t payload[64];
917 uint8_t len = 1;
918 payload[0] = cmdCode;
919
920 if (cmdCode == CMD_SET_PERIOD || cmdCode == CMD_WATER_TIME)
921 {
922 payload[len++] = (uint8_t)(argU16 & 0xFF);
923 payload[len++] = (uint8_t)((argU16 >> 8) & 0xFF);
924 }
925 else if (cmdCode == CMD_TEXT_MESSAGE && textPayload != nullptr)
926 {
927 // Slave rxBuffer = 64 bytes; max frame = 64: header(4) + cmdCode(1) + text(57) + CRC(2)
928 static constexpr uint8_t MAX_TEXT_LEN = 57;
929 uint8_t textLen = (uint8_t)strlen(textPayload);
930 if (textLen > MAX_TEXT_LEN)
931 {
932 textLen = MAX_TEXT_LEN;
933 }
934 memcpy(&payload[1], textPayload, textLen);
935 len += textLen;
936 }
937
938 uint8_t totalLen = buildFrame(FRAME_CMD, nodeId, seq, payload, len);
939 printTsv("CMD_TX", nodeId, seq, 0, 0, 0, 0, cmdCode, "send");
940 sendFrame(totalLen);
941 _masterState = STATE_WAIT_CMD_TXDONE;
942}

944void LoRaMaster::processHelloFrame()
945{
946 if (_rxSize < 7)
947 {
948 printTsv("HELLO_ERR", 0, 0, 0, 0, _lastRxRssi, _lastRxSnr, 0, "too_short");
949 startContinuousRx();
950 return;
951 }
952
953 uint8_t nodeId = _rxBuffer[1];
954 uint8_t seq = _rxBuffer[2];
955 uint8_t payloadLen = _rxBuffer[3];
956
957 if (payloadLen < 1)
958 {
959 printTsv("HELLO_ERR", nodeId, seq, 0, 0, _lastRxRssi, _lastRxSnr, 0, "bad_len");
960 startContinuousRx();
961 return;
962 }
963
964 uint8_t helloFlags = _rxBuffer[4];
965 bool bootFlag = (helloFlags & 0x80) != 0;
966 uint8_t statusFlag = (helloFlags & 0x01) ? 1 : 0;
967
968 updateNodeHello(nodeId, seq, statusFlag, _lastRxRssi, _lastRxSnr);
969 printTsv("HELLO_RX", nodeId, seq, 0, statusFlag, _lastRxRssi, _lastRxSnr, 0, "ok");
970 if (bootFlag)
971 {
972 publishNodeBootResult(nodeId);
973 }
974 clearPendingContext();
975
976 PendingNodeCommand *queued = getQueuedCommand(nodeId);
977 if (queued != nullptr && queued->active && queued->awaitingAck)
978 {
979 if (queued->retriesDone >= queued->maxRetries)
980 {
981 printTsv("CMD_DROP", nodeId, seq, 0, statusFlag, _lastRxRssi, _lastRxSnr, queued->cmdCode, "queued_max_retries");
982 publishQueuedCommandFailure(nodeId, queued->cmdCode, queued->argU16);
983 clearQueuedCommand(nodeId);
984 }
985 else
986 {
987 _pendingCmdAfterAck = true;
988 _pendingCmdTargetNode = nodeId;
989 _pendingCmdSeq = seq;
990 _pendingCmdCode = queued->cmdCode;
991 _pendingCmdArgU16 = queued->argU16;
992 strncpy(_pendingCmdText, queued->text, sizeof(_pendingCmdText) - 1);
993 _pendingCmdText[sizeof(_pendingCmdText) - 1] = '\0';
994 _pendingCmdIsQueued = true;
995 printTsv("CMD_WINDOW", nodeId, seq, 0, statusFlag, _lastRxRssi, _lastRxSnr, queued->cmdCode, queued->mqttVerb);
996 }
997 }
998 else if (hasScheduleWorkForNode(nodeId))
999 {
1000 _pendingCmdAfterAck = true;
1001 _pendingCmdTargetNode = nodeId;
1002 _pendingCmdSeq = seq;
1003 _pendingCmdCode = CMD_BATTERY;
1004 _pendingCmdArgU16 = 0;
1005 _pendingCmdText[0] = '\0';
1006 _pendingCmdIsScheduled = true;
1007 printTsv("SCHED_WINDOW", nodeId, seq, 0, statusFlag, _lastRxRssi, _lastRxSnr, CMD_BATTERY, "battery_request");
1008 }
1009
1010 sendAckHello(nodeId, seq);
1011}

1013void LoRaMaster::processAckCmdFrame()
1014{
1015 if (_rxSize < 3)
1016 {
1017 printTsv("ACK_CMD_ERR", 0, 0, 0, 0, _lastRxRssi, _lastRxSnr, 0, "too_short_hdr");
1018 startContinuousRx();
1019 return;
1020 }
1021
1022 uint8_t nodeId = _rxBuffer[1];
1023 uint8_t seq = _rxBuffer[2];
1024
1025 if (_rxSize < 8)
1026 {
1027 printTsv("ACK_CMD_ERR", nodeId, seq, 0, 0, _lastRxRssi, _lastRxSnr, 0, "too_short");
1028 if (_awaitingAckNodeId != 0)
1029 {
1030 PendingNodeCommand *queued = getQueuedCommand(_awaitingAckNodeId);
1031 if (queued != nullptr && queued->active)
1032 {
1033 publishQueuedCommandFailure(_awaitingAckNodeId, queued->cmdCode, queued->argU16);
1034 clearQueuedCommand(_awaitingAckNodeId);
1035 }
1036 }
1037 _awaitingAckNodeId = 0;
1038 _awaitingAckSeq = 0;
1039 _awaitingAckCmd = CMD_NONE;
1040 _awaitingAckStartedMs = 0;
1041 startContinuousRx();
1042 return;
1043 }
1044
1045 uint8_t payloadLen = _rxBuffer[3];
1046 uint8_t ackedCmd = _rxBuffer[4];
1047 uint8_t ackStatus = _rxBuffer[5];
1048 bool ok = (ackStatus == ACK_STATUS_OK);
1049
1050 PendingNodeCommand *queued = getQueuedCommand(nodeId);
1051
1052 if (ackedCmd == CMD_BATTERY)
1053 {
1054 if (ok && payloadLen >= 4)
1055 {
1056 uint16_t batteryMv = (uint16_t)_rxBuffer[6] | ((uint16_t)_rxBuffer[7] << 8);
1057 updateNodeBattery(nodeId, batteryMv);
1058 printTsv("ACK_CMD_RX", nodeId, seq, batteryMv, 0, _lastRxRssi, _lastRxSnr, ackedCmd, "battery_ok");
1059 publishBatteryResult(nodeId, true, batteryMv, 0, _lastRxRssi, _lastRxSnr);
1060 }
1061 else if (!_pendingCmdIsScheduled)
1062 {
1063 publishBatteryResult(nodeId, false, 0, 0, 0, 0);
1064 }
1065
1066 if (_schedule.cycleActive && _awaitingAckNodeId == nodeId && _awaitingAckSeq == seq && _awaitingAckCmd == CMD_BATTERY)
1067 {
1068 markScheduleDoneForNode(nodeId);
1069 }
1070 }
1071 else if (ackedCmd == CMD_STATUS)
1072 {
1073 if (ok && payloadLen >= 9)
1074 {
1075 uint16_t batteryMv = (uint16_t)_rxBuffer[6] | ((uint16_t)_rxBuffer[7] << 8);
1076 uint8_t statusFlag = _rxBuffer[8];
1077 uint32_t uptimeSec = (uint32_t)_rxBuffer[9] |
1078 ((uint32_t)_rxBuffer[10] << 8) |
1079 ((uint32_t)_rxBuffer[11] << 16) |
1080 ((uint32_t)_rxBuffer[12] << 24);
1081 updateNodeStatus(nodeId, batteryMv, statusFlag, uptimeSec, (uint16_t)abs(_lastRxRssi), _lastRxSnr);
1082 printTsv("ACK_CMD_RX", nodeId, seq, batteryMv, statusFlag, _lastRxRssi, _lastRxSnr, ackedCmd, "status_ok");
1083 publishStatusResult(nodeId, true, batteryMv, statusFlag, uptimeSec, _lastRxRssi, _lastRxSnr);
1084 }
1085 else
1086 {
1087 publishStatusResult(nodeId, false, 0, 0, 0, 0, 0);
1088 }
1089 }
1090 else if (ackedCmd == CMD_WATER_STATUS)
1091 {
1092 if (ok && payloadLen >= 19)
1093 {
1094 uint8_t valveOpen = _rxBuffer[6];
1095 uint8_t wateringActive = _rxBuffer[7];
1096 uint16_t durationSec = (uint16_t)_rxBuffer[8] | ((uint16_t)_rxBuffer[9] << 8);
1097 uint16_t remainingSec = (uint16_t)_rxBuffer[10] | ((uint16_t)_rxBuffer[11] << 8);
1098
1099 uint32_t flowPulses = (uint32_t)_rxBuffer[12] |
1100 ((uint32_t)_rxBuffer[13] << 8) |
1101 ((uint32_t)_rxBuffer[14] << 16) |
1102 ((uint32_t)_rxBuffer[15] << 24);
1103
1104 uint16_t litres = (uint16_t)_rxBuffer[16] | ((uint16_t)_rxBuffer[17] << 8);
1105 uint32_t postClosePulses = (uint32_t)_rxBuffer[18] |
1106 ((uint32_t)_rxBuffer[19] << 8) |
1107 ((uint32_t)_rxBuffer[20] << 16) |
1108 ((uint32_t)_rxBuffer[21] << 24);
1109
1110 uint8_t postCloseLeakDetected = _rxBuffer[22];
1111
1112 updateNodeWaterStatus(nodeId, valveOpen, wateringActive, durationSec, remainingSec, flowPulses, litres);
1113 printTsv("ACK_CMD_RX", nodeId, seq, 0, 0, _lastRxRssi, _lastRxSnr, ackedCmd, "wstatus_ok");
1114 publishWaterStatusResult(nodeId, true, valveOpen, wateringActive,
1115 durationSec, remainingSec, flowPulses, litres,
1116 postClosePulses, postCloseLeakDetected);
1117 }
1118 else
1119 {
1120 publishWaterStatusResult(nodeId, false, 0, 0, 0, 0, 0, 0, 0, 0);
1121 }
1122 }
1123 else if (ackedCmd == CMD_PING)
1124 {
1125 publishPingResult(nodeId, ok);
1126 }
1127 else if (ackedCmd == CMD_TEXT_MESSAGE)
1128 {
1129 publishTextResult(nodeId, ok);
1130 }
1131 else if (ackedCmd == CMD_REBOOT)
1132 {
1133 publishRebootResult(nodeId, ok);
1134 }
1135 else if (ackedCmd == CMD_SET_PERIOD)
1136 {
1137 uint16_t seconds = (queued != nullptr) ? queued->argU16 : 0;
1138 publishPeriodResult(nodeId, ok, seconds);
1139 }
1140 else if (ackedCmd == CMD_LED_ON || ackedCmd == CMD_LED_OFF)
1141 {
1142 uint8_t value = (ackedCmd == CMD_LED_ON) ? 1 : 0;
1143 publishLedResult(nodeId, ok, value);
1144 }
1145 else if (ackedCmd == CMD_VALVE_OPEN || ackedCmd == CMD_VALVE_CLOSE)
1146 {
1147 uint8_t value = (ackedCmd == CMD_VALVE_OPEN) ? 1 : 0;
1148 publishValveResult(nodeId, ok, value);
1149 }
1150 else if (ackedCmd == CMD_WATER_TIME)
1151 {
1152 if (ok && payloadLen >= 19)
1153 {
1154 uint8_t valveOpen = _rxBuffer[6];
1155 uint8_t wateringActive = _rxBuffer[7];
1156 uint16_t durationSec = (uint16_t)_rxBuffer[8] | ((uint16_t)_rxBuffer[9] << 8);
1157 uint16_t remainingSec = (uint16_t)_rxBuffer[10] | ((uint16_t)_rxBuffer[11] << 8);
1158
1159 uint32_t flowPulses = (uint32_t)_rxBuffer[12] |
1160 ((uint32_t)_rxBuffer[13] << 8) |
1161 ((uint32_t)_rxBuffer[14] << 16) |
1162 ((uint32_t)_rxBuffer[15] << 24);
1163
1164 uint16_t litres = (uint16_t)_rxBuffer[16] | ((uint16_t)_rxBuffer[17] << 8);
1165
1166 uint32_t postClosePulses = (uint32_t)_rxBuffer[18] |
1167 ((uint32_t)_rxBuffer[19] << 8) |
1168 ((uint32_t)_rxBuffer[20] << 16) |
1169 ((uint32_t)_rxBuffer[21] << 24);
1170
1171 uint8_t postCloseLeakDetected = _rxBuffer[22];
1172
1173 updateNodeWaterStatus(nodeId, valveOpen, wateringActive, durationSec, remainingSec, flowPulses, litres);
1174 printTsv("ACK_CMD_RX", nodeId, seq, 0, 0, _lastRxRssi, _lastRxSnr, ackedCmd, "water_ok");
1175 publishWaterStartResult(nodeId, true, valveOpen, wateringActive,
1176 durationSec, remainingSec, flowPulses, litres,
1177 postClosePulses, postCloseLeakDetected);
1178 }
1179 else
1180 {
1181 publishWaterStartResult(nodeId, false, 0, 0, 0, 0, 0, 0, 0, 0);
1182 }
1183 }
1184
1185 else if (ackedCmd == CMD_WATER_ABORT)
1186 {
1187 if (ok && payloadLen >= 19)
1188 {
1189 uint8_t valveOpen = _rxBuffer[6];
1190 uint8_t wateringActive = _rxBuffer[7];
1191 uint16_t durationSec = (uint16_t)_rxBuffer[8] | ((uint16_t)_rxBuffer[9] << 8);
1192 uint16_t remainingSec = (uint16_t)_rxBuffer[10] | ((uint16_t)_rxBuffer[11] << 8);
1193 uint32_t flowPulses = (uint32_t)_rxBuffer[12] |
1194 ((uint32_t)_rxBuffer[13] << 8) |
1195 ((uint32_t)_rxBuffer[14] << 16) |
1196 ((uint32_t)_rxBuffer[15] << 24);
1197
1198 uint16_t litres = (uint16_t)_rxBuffer[16] | ((uint16_t)_rxBuffer[17] << 8);
1199
1200 uint32_t postClosePulses = (uint32_t)_rxBuffer[18] |
1201 ((uint32_t)_rxBuffer[19] << 8) |
1202 ((uint32_t)_rxBuffer[20] << 16) |
1203 ((uint32_t)_rxBuffer[21] << 24);
1204
1205 uint8_t postCloseLeakDetected = _rxBuffer[22];
1206 updateNodeWaterStatus(nodeId, valveOpen, wateringActive, durationSec, remainingSec, flowPulses, litres);
1207 printTsv("ACK_CMD_RX", nodeId, seq, 0, 0, _lastRxRssi, _lastRxSnr, ackedCmd, "abort_ok");
1208 publishWaterAbortResult(nodeId, true, valveOpen, wateringActive,
1209 durationSec, remainingSec, flowPulses, litres,
1210 postClosePulses, postCloseLeakDetected);
1211 }
1212 else
1213 {
1214 publishWaterAbortResult(nodeId, false, 0, 0, 0, 0, 0, 0, 0, 0);
1215 }
1216 }
1217 else if (ackedCmd == CMD_OTA_MODE)
1218 {
1219 publishNodeOtaResult(nodeId, ok);
1220 }
1221
1222 if (queued != nullptr && queued->active && queued->awaitingAck && seq == queued->lastSeq && ackedCmd == queued->cmdCode)
1223 {
1224 clearQueuedCommand(nodeId);
1225 }
1226
1227 if (_awaitingAckNodeId == nodeId && _awaitingAckSeq == seq && _awaitingAckCmd == ackedCmd)
1228 {
1229 _awaitingAckNodeId = 0;
1230 _awaitingAckSeq = 0;
1231 _awaitingAckCmd = CMD_NONE;
1232 _awaitingAckStartedMs = 0;
1233 }
1234
1235 if (ok)
1236 {
1237 publishNodeSeen(nodeId);
1238 }
1239
1240 startContinuousRx();
1241}

1243void LoRaMaster::processReceivedFrame()
1244{
1245 if (_rxSize < 6)
1246 {
1247 printTsv("RX_ERR", 0, 0, 0, 0, _lastRxRssi, _lastRxSnr, 0, "frame_too_short");
1248 startContinuousRx();
1249 return;
1250 }
1251
1252 if (!validateReceivedFrame())
1253 {
1254 printTsv("CRC_ERR", 0, 0, 0, 0, _lastRxRssi, _lastRxSnr, 0, "bad_frame");
1255 startContinuousRx();
1256 return;
1257 }
1258
1259 printFrameDebug("RX", _rxBuffer, (uint8_t)_rxSize);
1260
1261 switch (_rxBuffer[0])
1262 {
1263 case FRAME_HELLO:
1264 processHelloFrame();
1265 break;
1266 case FRAME_ACK_CMD:
1267 processAckCmdFrame();
1268 break;
1269 default:
1270 printTsv("RX_UNKNOWN", _rxBuffer[1], _rxBuffer[2], 0, 0, _lastRxRssi, _lastRxSnr, _rxBuffer[0], "unsupported");
1271 startContinuousRx();
1272 break;
1273 }
