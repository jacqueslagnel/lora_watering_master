204void loop()
205{
206 // ------------ OTA mode ---------------------
207 if (BleOta::isActive())
208 {
209 BleOta::loop();

211 __WFI();

213 }

217
218 if (PC_SERIAL.available())

220 String line = PC_SERIAL.readStringUntil('\n');

222

224 lower.toLowerCase();
225
226 if (lower == "pwr" ||

229 lower.startsWith("ka ") ||
230 lower.startsWith("at"))
231 {
232 lte_modem.handleConsoleLine(line);

238 }

240 lte_modem.loop();
241 // @@@@ envoie mqtt on connect
242 if (lte_modem.consumeJustConnected())
243 {
244 lte_modem.publish("LTE_CONNECTED;" + lte_modem.getLteStatus());
245 lora.republishAllNodePresence();

247 processIncomingMqtt();

275static float readVBAT(void)
276{
277 uint32_t acc = 0;
278 const uint8_t n = 8;
279 for (uint8_t i = 0; i < n; i++)
280 {
281 acc += (uint32_t)analogRead(vbat_pin);
282 delay(2);
283 }
284 float raw = (float)acc / n;
285 return raw * REAL_VBAT_MV_PER_LSB;
286}
