201void Sim7080Mqtt::handleModemLine(const String &line)
202{
203 if (line.length() == 0)
204 {
205 return;
206 }
207
208 _pc.print("[MODEM] ");
209 _pc.println(line);
210
211 if (line.indexOf("+CEREG:") >= 0)
212 {
213 int comma = line.indexOf(',');
214 bool registered = false;
215
216 if (comma >= 0)
217 {
218 char stat = line.charAt(comma + 1);
219 if (stat == '1' || stat == '5')
220 {
221 char after = (comma + 2 < (int)line.length()) ? line.charAt(comma + 2) : '\n';
222 if (after == ',' || after == '\r' || after == '\n' || after == '\0' || after == ' ')
223 {
224 registered = true;
225 }
226 }
227 }
228
229 _networkRegistered = registered;
230 printStateSnapshot("CEREG");
231 }
232
233 if (line.indexOf("+CGATT: 1") >= 0)
234 {
235 _packetAttached = true;
236 printStateSnapshot("CGATT");
237 }
238 else if (line.indexOf("+CGATT: 0") >= 0)
239 {
240 _packetAttached = false;
241 printStateSnapshot("CGATT");
242 }
243
244 // +SMSTATE: 0 = disconnected URC, +SMSTATE: 1 = connected URC
245 if (line.startsWith("+SMSTATE:"))
246 {
247 int colonPos = line.indexOf(':');
248 if (colonPos >= 0)
249 {
250 String val = line.substring(colonPos + 1);
251 val.trim();
252 if (val == "0")
253 {
254 _mqttConnected = false;
255 printStateSnapshot("SMSTATE DISC");
256 }
257 else if (val == "1")
258 {
259 _mqttConnected = true;
260 _lastMqttOkMs = millis();
261 printStateSnapshot("SMSTATE CONN");
262 }
263 }
264 }
265
266 // +SMSUB: "topic","payload" — single-line URC per SIM7080G app note §5.1
267 if (line.startsWith("+SMSUB:"))
268 {
269 _lastMqttOkMs = millis();
270 // Extract payload: last ,"<payload>" portion
271 int splitPos = line.lastIndexOf(",\"");
272 if (splitPos >= 0)
273 {
274 String payload = line.substring(splitPos + 2);
275 if (payload.length() > 0 && payload.charAt(payload.length() - 1) == '"')
276 {
277 payload = payload.substring(0, payload.length() - 1);
278 }
279 if (_mqttQueueCount < MQTT_QUEUE_SIZE)
280 {
281 _mqttQueue[_mqttQueueTail] = payload;
282 _mqttQueueTail = (_mqttQueueTail + 1) % MQTT_QUEUE_SIZE;
283 _mqttQueueCount++;
284 }
285 else
286 {
287 logWarn("MQTT queue pleine, message perdu");
288 }
289 }
290 }

575bool Sim7080Mqtt::tryNetworkProfile(const char *name, const char *plmn, int act, int cmnb)
576{
577 logInfo(String("Essai profil LTE: ") + name);
578
579 _networkRegistered = false;
580 _packetAttached = false;
581
582 {
583 // AT+CMNB: 1=LTE-M, 2=NB-IoT, 3=both (replaces BG77 AT+QCFG="iotopmode")
584 String cmd = "AT+CMNB=";

592 }
593

596 cmd += plmn;
597 cmd += "\",";
598 cmd += String(act);
599
600 if (!sendATOK(cmd, 30000))
601 {
602 logWarn(String("COPS refuse pour ") + name);

606
607 if (!waitForRegistration(45000))

609 logWarn(String("Pas d'enregistrement reseau pour ") + name);

614 _lteStatus.replace(" NB-IoT", ";NB-IoT");
615 _lteStatus.replace(" Cat-M1", ";Cat-M1");
616
617 logInfo(String("Profil LTE OK: ") + name);
618
619 // Signal quality via AT+CPSI? (replaces BG77 AT+QCSQ)
620 // Response: +CPSI: LTE NB,Online,MCC-MNC,TAC,CID,BAND,EARFCN,...,RSRP,RSRQ,SINR,...
621 String cpsiResp;
622 sendATWaitFor("AT+CPSI?", "+CPSI:", nullptr, 5000, &cpsiResp);
623 int cpsiIdx = cpsiResp.indexOf("+CPSI:");
624 if (cpsiIdx >= 0)
625 {
626 String cpsiLine = cpsiResp.substring(cpsiIdx + 6);
627 cpsiLine.trim();
628 // Skip first two fields (type and "Online") to get signal data
629 int c1 = cpsiLine.indexOf(',');

631 if (c2 >= 0)
632 {
633 String sigPart = cpsiLine.substring(c2 + 1);
634 sigPart.replace(",", ";");
635 if (sigPart.length() > 0)
636 {
637 _lteStatus += ";";
638 _lteStatus += sigPart;
639 }
640 }
641 }
642
643 sendATWaitFor("AT+COPS?", "+COPS:", nullptr, 5000, nullptr);

645 return true;

765bool Sim7080Mqtt::publishToTopic(const String &topic, const String &payload)
766{
767 if (!_mqttConnected)
768 {
769 return false;
770 }
771
772 // AT+SMPUB uses a two-phase protocol: send command, wait for '>' prompt,
773 // then send payload + Ctrl-Z and wait for OK.
774 String cmd = "AT+SMPUB=\"";
775 cmd += topic;
776 cmd += "\",";
777 cmd += String(payload.length());
778 cmd += ",0,0";
779
780 _pc.print(">> ");
781 _pc.println(cmd);
782 clearModemInput();
783 _modem.println(cmd);
784
785 // Phase 1: wait for '>' prompt (character-level, not line-based)
786 uint32_t start = millis();
787 bool gotPrompt = false;
788 while ((uint32_t)(millis() - start) < 10000UL && !gotPrompt)
789 {
790 watchdogFeed();
791 while (_modem.available())
792 {
793 char c = (char)_modem.read();
794 if (c == '>')
795 {
796 gotPrompt = true;
797 break;
798 }
799 }
800 if (!gotPrompt)
801 delay(1);
802 }
803
804 if (!gotPrompt)
805 {
806 logError("SMPUB: pas de prompt '>'");
807 markMqttLost("SMPUB no prompt");
808 return false;
809 }
810
811 // Phase 2: send exactly payload.length() bytes — SIM7080G is length-terminated, no Ctrl-Z
812 _modem.print(payload);
813
814 // Phase 3: wait for OK or ERROR
815 start = millis();
816 while ((uint32_t)(millis() - start) < 15000UL)
817 {
818 watchdogFeed();
819 while (_modem.available())
820 {
821 char c = (char)_modem.read();
822 if (c == '\r')
823 continue;
824 if (c == '\n')
825 {
826 _modemLine.trim();
827 if (_modemLine.length() > 0)
828 {
829 handleModemLine(_modemLine);
830 if (_modemLine.indexOf("OK") >= 0)
831 {
832 _modemLine = "";
833 _lastMqttOkMs = millis();
834 return true;
835 }
836 if (_modemLine.indexOf("ERROR") >= 0)
837 {
838 _modemLine = "";
839 markMqttLost("SMPUB publish failed");
840 return false;
841 }
842 }
843 _modemLine = "";
844 }
845 else
846 {
847 _modemLine += c;
848 }
849 }
850 delay(1);
851 }
852
853 markMqttLost("SMPUB timeout");
854 return false;
855}

880uint32_t Sim7080Mqtt::reconnectBackoffMs() const
881{
882 if (_mqttReconnectFailures == 0)
883 {
884 return MQTT_RETRY_DELAY_MS;
885 }
886
887 uint32_t backoff = MQTT_RETRY_DELAY_MS;
888 for (uint8_t i = 0; i < _mqttReconnectFailures; i++)
889 {
890 backoff *= 2UL;
891 if (backoff > 120000UL)
892 {
893 backoff = 120000UL;
894 break;
895 }
896 }
897 return backoff;
898}

900bool Sim7080Mqtt::init()
901{
902 logInfo("Init complete LTE fallback + MQTT SIM7080G");
903
904 if (!_bootPowerCycleDone)
905 {
906 logWarn("Power cycle SIM7080G au boot");
907 modemPowerCycle();
908 _bootPowerCycleDone = true;
909 }
910
911 if (!prepareModemAfterPowerOn())
912 {
913 logError("SIM7080G non pret");
914 return false;
915 }
916
917 updateRegistration();
918
919 if (!_networkRegistered)
920 {
921 sendATWaitFor("ATI", "OK", "ERROR", 5000, nullptr);
922
923 bool registered = false;
924
925 // cmnb: 2=NB-IoT, 1=LTE-M (replaces BG77 iotopmode 1=NB-IoT, 0=Cat-M1)
926 registered = tryNetworkProfile("Bouygues NB-IoT", PLMN_BOUYGUES, ACT_NBIOT, 2);
927
928 if (!registered)
929 {
930 logWarn("Echec Bouygues NB-IoT -> power cycle SIM7080G");
931 modemPowerCycle();
932 }
933
934 if (!registered)
935 registered = tryNetworkProfile("Orange NB-IoT", PLMN_ORANGE, ACT_NBIOT, 2);
936
937 if (!registered)
938 {
939 logWarn("Echec Orange NB-IoT -> power cycle SIM7080G");
940 modemPowerCycle();
941 }
942
943 if (!registered)
944 registered = tryNetworkProfile("Bouygues Cat-M1", PLMN_BOUYGUES, ACT_CATM1, 1);
945
946 if (!registered)
947 {
948 logWarn("Echec Bouygues Cat-M1 -> power cycle SIM7080G");
949 modemPowerCycle();
950 }
951
952 if (!registered)
953 registered = tryNetworkProfile("Orange Cat-M1", PLMN_ORANGE, ACT_CATM1, 1);
954
955 if (!registered)
956 {
957 logWarn("Echec Orange Cat-M1 -> power cycle SIM7080G");
958 modemPowerCycle();
959 logError("Aucun profil LTE disponible");
960 return false;
961 }
962 }
963 else
964 {
965 logInfo("Deja enregistre, reconnect direct");
966
967 String cpsiResp;
968 sendATWaitFor("AT+CPSI?", "+CPSI:", nullptr, 5000, &cpsiResp);
969 int cpsiIdx = cpsiResp.indexOf("+CPSI:");
970 if (cpsiIdx >= 0)
971 {
972 String cpsiLine = cpsiResp.substring(cpsiIdx + 6);
973 cpsiLine.trim();
974 int c1 = cpsiLine.indexOf(',');
975 int c2 = (c1 >= 0) ? cpsiLine.indexOf(',', c1 + 1) : -1;
976 if (c2 >= 0)
977 {
978 int s1 = _lteStatus.indexOf(';');
979 int s2 = (s1 >= 0) ? _lteStatus.indexOf(';', s1 + 1) : -1;
980 String base = (s2 >= 0) ? _lteStatus.substring(0, s2) : _lteStatus;
981 String sigPart = cpsiLine.substring(c2 + 1);
982 sigPart.replace(",", ";");
983 if (sigPart.length() > 0)
984 _lteStatus = base + ";" + sigPart;
985 }
986 }
987
988 sendATWaitFor("AT+COPS?", "+COPS:", nullptr, 5000, nullptr);
989 }
990
991 {
992 // AT+CGDCONT: standard 3GPP APN config (also present in SIM7080G APN manual config §4.2)
993 String cmd = "AT+CGDCONT=1,\"IP\",\"";
994 cmd += APN_NAME;
995 cmd += "\"";
996 if (!sendATOK(cmd, 5000))
997 {
998 logError("CGDCONT refuse");
999 return false;
1000 }
1001 }
1002
1003 if (!updatePacketAttach())
1004 {
1005 logError("CGATT? echec");
1006 return false;
1007 }
1008
1009 if (!_packetAttached)
1010 {
1011 if (!sendATOK("AT+CGATT=1", 30000))
1012 {
1013 logError("CGATT=1 refuse");
1014 return false;
1015 }
1016
1017 if (!updatePacketAttach())
1018 {
1019 logError("CGATT? apres attach echec");
1020 return false;
1021 }
1022
1023 if (!_packetAttached)
1024 {
1025 logError("Packet attach absent");
1026 return false;
1027 }
1028 }
1029
1030 // AT+CNCFG + AT+CNACT: SIM7080G PDP context activation (mandatory, replaces BG77 CGPADDR)
1031 {
1032 String cmd = "AT+CNCFG=0,1,\"";
1033 cmd += APN_NAME;
1034 cmd += "\"";
1035 sendATOK(cmd, 5000); // best-effort: may already be configured
1036 }
1037
1038 // Activate PDP context 0; OK comes immediately, +APP PDP: 0,ACTIVE follows as URC.
1039 // If the context is stuck in an old state, deactivate once and retry before failing.
1040 if (!sendATOK("AT+CNACT=0,1", 30000))
1041 {
1042 logWarn("CNACT=0,1 refuse, reset contexte PDP");
1043 sendATOK("AT+CNACT=0,0", 15000);
1044 delay(1000);
1045 if (!sendATOK("AT+CNACT=0,1", 30000))
1046 {
1047 logError("CNACT=0,1 echec");
1048 return false;
1049 }
1050 }
1051 pumpModem(2000); // drain +APP PDP: 0,ACTIVE URC
1052
1053 if (!hasIPAddress())
1054 {
1055 logError("Pas d'adresse IP");
1056 return false;
1057 }
1058
1059 return mqttReconnect();
1060}
